set(NV "${CMAKE_CURRENT_SOURCE_DIR}/../../src/nv2a/nv2a_pgraph_d3d11.c")
file(READ "${NV}" content)
function(extract first last output)
  string(FIND "${content}" "${first}" begin)
  string(FIND "${content}" "${last}" end)
  if(begin LESS 0 OR end LESS begin)
    message(FATAL_ERROR "HUD layout extraction boundary changed")
  endif()
  math(EXPR length "${end}-${begin}")
  string(SUBSTRING "${content}" ${begin} ${length} part)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${part}")
endfunction()
extract("static int nv_menu_surface(" "static int nv_timer_body_edge(" clip.h)
extract("static int nv_hud_geometry(" "static void nv_sync_render_target(" layout.h)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${NV}")
add_executable(hud_layout_test hud_layout_test.c ../../src/game/video_settings.c)
target_include_directories(hud_layout_test PRIVATE ../../src/game ../../src/game/recomp/gen "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(hud_layout_test PRIVATE user32)
add_test(NAME hud_layout COMMAND hud_layout_test)

# Exercise the exact runtime resize transaction without a graphics device.
set(DEVICE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/d3d/d3d8_device.c")
file(READ "${DEVICE}" device_content)
string(FIND "${device_content}" "static void d3d8_apply_target_size_change(void)" resize_begin)
string(FIND "${device_content}" "static void d3d8_keep_prev_frame(void);" resize_end)
if(resize_begin LESS 0 OR resize_end LESS resize_begin)
    message(FATAL_ERROR "Resize transaction extraction boundary changed")
endif()
math(EXPR resize_length "${resize_end}-${resize_begin}")
string(SUBSTRING "${device_content}" ${resize_begin} ${resize_length} resize_code)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/target_resize_impl.h" "${resize_code}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${DEVICE}")
add_executable(target_resize_test target_resize_test.c ../../src/game/video_settings.c)
target_include_directories(target_resize_test PRIVATE ../../src/game ../../src/game/recomp/gen "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(target_resize_test PRIVATE user32)
add_test(NAME target_resize COMMAND target_resize_test)
